== cost.sh
#!/bin/bash
# Builds old and new writers, then times MC chr22 three times each, alternating
cd ~/work/tidy7
for v in old new; do (cd $v && cargo build --release --locked -q) || exit 1; done
IN=~/work/arb-format-opus/in
cap() { systemd-run --user --scope -q -p MemoryMax=100G -p MemorySwapMax=0 "$@"; }
for k in 1 2 3; do
  for v in old new; do
    cap /usr/bin/time -f "$v run $k: %e s, %M KB peak" $v/target/release/gfa-to-tabix $IN/mc22.gfa --walks --refs GRCh38 -o $v.mc22.$k > $v.mc22.$k.log 2>&1
    tail -1 $v.mc22.$k.log >> cost.txt
  done
done
for v in old new; do
  cap /usr/bin/time -f "$v pggbY: %e s, %M KB peak" $v/target/release/gfa-to-tabix $IN/pggbY.gfa --walks --refs GRCh38 -o $v.pggbY > $v.pggbY.log 2>&1
  tail -1 $v.pggbY.log >> cost.txt
done
echo done >> cost.txt
